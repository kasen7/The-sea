滑动窗口-长度最小的子数组
{
        int size=nums.size();
        int left,right,len,sum;
        left=right=len=sum=0;

        while(right<size)
        {
            sum+=nums[right];
            while(sum>=target)
            {
                int tmp=right-left+1;
                if(len==0)
                    len=tmp;
                else
                    len=min(tmp,len);
                sum-=nums[left++];
            }
            right++;
        }
        return len;
}
